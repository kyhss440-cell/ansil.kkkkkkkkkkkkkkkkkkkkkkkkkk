  function $(id) { return document.getElementById(id); }
  function esc(s) { return String(s).replace(/[&<>"']/g, function (c) { return { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]; }); }

  function toast(msg, kind) {
    kind = kind || "ok";
    const el = document.createElement("div");
    el.className = "toast " + kind;
    el.innerHTML = '<span class="dot"></span><span>' + esc(msg) + '</span>';
    $("toasts").appendChild(el);
    setTimeout(function () { el.classList.add("hide"); setTimeout(function () { el.remove(); }, 320); }, 2400);
  }

  function updateStreak() {
    const today = new Date().toDateString();
    const st = state.progress.streak;
    if (st.last === today) return;
    const yesterday = new Date(Date.now() - 86400000).toDateString();
    if (st.last === yesterday) st.count += 1; else st.count = 1;
    st.last = today; saveProgress();
  }
  function markActivity() {
    const key = new Date().toISOString().slice(0, 10);
    state.progress.activity[key] = (state.progress.activity[key] || 0) + 1;
    saveProgress();
  }
  function chapterProgress(subjId, chId) {
    const rec = state.progress.chapters[state.classLevel + ":" + subjId + ":" + chId];
    if (!rec) return 0;
    return Math.min(1, (rec.best || 0) / 100);
  }
  function subjectProgress(subjId) {
    const s = findSubject(subjId);
    if (!s || !s.chapters.length) return 0;
    let sum = 0;
    s.chapters.forEach(function (c) { sum += chapterProgress(subjId, c.id); });
    return sum / s.chapters.length;
  }

    let saved = null;
    try { saved = localStorage.getItem("sm-theme"); } catch (e) {}

    else { const dark = window.matchMedia && window.matchMedia("(prefers-color-scheme: dark)").matches; applyTheme(dark ? "dark" : "light"); }

    applyTheme(now === "dark" ? "light" : "dark");
  });

  function applyClass(level) {

    const label = level === "8" ? "Class 8 · Dark Green" : "Class " + level + " · Smart Learning";
    $("sbClassLabel").textContent = label;
    try { localStorage.setItem("sm-class", level); } catch (e) {}

  function computeOverallBest() {
    const qs = state.progress.quizzes.filter(function (q) { return !q.classLevel || q.classLevel === state.classLevel; });
    if (!qs.length) return 0;
    let best = 0;
    qs.forEach(function (q) { if (q.score > best) best = q.score; });
    return Math.round(best);
  }

  function startQuiz(subjId, chId, length) {
    const s = findSubject(subjId);
    const c = findChapter(subjId, chId);
    if (!s || !c) return;
    const n = length || c.q.length;
    let pool = c.q.slice();
    shuffle(pool);
    const questions = pool.slice(0, Math.min(n, pool.length));
    state.quiz = {
      active: true, finished: false, subjectId: subjId, chapterId: chId,
      subjectName: s.name, chapterName: c.name, questions: questions,
      index: 0, score: 0, answers: [], revealed: false
    };
    setView("quiz");
  }

  function shuffle(arr) {
    for (let i = arr.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      const tmp = arr[i]; arr[i] = arr[j]; arr[j] = tmp;
    }
  }

  function renderQuizActive() {
    const q = state.quiz;
    if (!q || !q.active) return;
    const total = q.questions.length;
    const cur = q.questions[q.index];
    if (!cur) return;
    const pct = (q.index / total) * 100;
    const letters = ["A", "B", "C", "D"];

    let html = '<div class="quiz-shell">';
    html += '<div class="quiz-top"><div class="info">' + esc(q.subjectName) + ' <span class="sub">· ' + esc(q.chapterName) + '</span></div>';
    html += '<div class="info">' + esc(t("question")) + ' ' + (q.index + 1) + ' / ' + total + '</div></div>';
    html += '<div class="qbar"><div class="qbar-fill" style="width:' + pct + '%"></div></div>';
    html += '<div class="qcard"><div class="qnum">' + esc(t("question")) + ' ' + (q.index + 1) + '</div>';
    html += '<div class="qtext">' + esc(cur.q) + '</div>';

    cur.o.forEach(function (opt, i) {
      let cls = "opt", mark = "";
      if (q.revealed) {
        cls += " locked";
        if (i === cur.a) { cls += " correct"; mark = "✓"; }
        else if (q.answers[q.index] === i) { cls += " wrong"; mark = "✗"; }
        else { cls += " dim"; }
      }
      html += '<button type="button" class="' + cls + '" data-opt="' + i + '" data-qindex="' + q.index + '">';
      html += '<span class="letter">' + letters[i] + '</span>';
      html += '<span class="opt-text">' + esc(opt) + '</span>';
      html += '<span class="mark">' + mark + '</span></button>';
    });

    if (q.revealed) {
      if (cur.e) { html += '<div class="explain-box"><div class="ex-title">' + esc(t("whyCorrect")) + '</div>' + esc(cur.e) + '</div>'; }
      else if (cur.h) { html += '<div class="hint-box"><strong>' + esc(t("hintLabel")) + '</strong> ' + esc(cur.h) + '</div>'; }
    }
    html += '</div>';

    html += '<div class="quiz-foot">';
    if (!q.revealed) {
      html += '<button class="btn btn-ghost" id="qHint">' + esc(t("showHint")) + '</button>';
      html += '<button class="btn btn-ghost" id="qReveal">' + esc(t("revealAnswer")) + '</button>';
    } else {
      if (q.index < total - 1) html += '<button class="btn btn-primary" id="qNext">' + esc(t("nextQuestion")) + '</button>';
      else html += '<button class="btn btn-primary" id="qFinish">' + esc(t("seeResults")) + '</button>';
    }
    html += '</div></div>';

    view.innerHTML = html;

    view.querySelectorAll(".opt").forEach(function (btn) {
      btn.addEventListener("click", function (e) {
        e.preventDefault(); e.stopPropagation();
        if (!state.quiz || !state.quiz.active || state.quiz.revealed) return;
        const btnQIndex = parseInt(btn.getAttribute("data-qindex"), 10);
        if (btnQIndex !== state.quiz.index) return;
        handleAnswer(parseInt(btn.getAttribute("data-opt"), 10));
      });
    });

    const hintBtn = view.querySelector("#qHint");
    if (hintBtn) hintBtn.addEventListener("click", function () {
      const c = state.quiz.questions[state.quiz.index];
      if (c.h) toast(c.h, "warn"); else toast("No hint for this one!", "warn");
    });

    if (revealBtn) revealBtn.addEventListener("click", function () {
      if (!state.quiz || state.quiz.revealed) return;
      state.quiz.answers[state.quiz.index] = -1;
      state.quiz.revealed = true;
      renderQuizActive();
    });

    if (nextBtn) nextBtn.addEventListener("click", function () { state.quiz.index++; state.quiz.revealed = false; renderQuizActive(); });
    const finBtn = view.querySelector("#qFinish");
    if (finBtn) finBtn.addEventListener("click", finishQuiz);
  }

  function handleAnswer(chosen) {
    const q = state.quiz;
    if (!q || !q.active || q.revealed) return;
    const cur = q.questions[q.index];
    if (!cur) return;
    q.answers[q.index] = chosen;
    q.revealed = true;
    if (chosen === cur.a) { q.score++; toast("Correct! 🎉", "ok"); }
    else if (chosen >= 0) toast("Not quite — correct answer highlighted.", "warn");
    renderQuizActive();
  }

  function finishQuiz() {
    const q = state.quiz;
    if (!q) return;
    const total = q.questions.length;
    const correct = q.score;
    const pct = Math.round((correct / total) * 100);
    state.progress.quizzes.push({
      classLevel: state.classLevel, subjectId: q.subjectId, subjectName: q.subjectName,
      chapterId: q.chapterId, chapterName: q.chapterName,
      score: pct, correct: correct, total: total, date: Date.now()
    });
    if (state.progress.quizzes.length > 80) state.progress.quizzes = state.progress.quizzes.slice(-80);
    const key = state.classLevel + ":" + q.subjectId + ":" + q.chapterId;
    const prev = state.progress.chapters[key] || { best: 0, attempts: 0 };
    state.progress.chapters[key] = { best: Math.max(prev.best || 0, pct), attempts: (prev.attempts || 0) + 1, last: pct };
    saveProgress(); markActivity();
    q.active = false; q.finished = true; q.finalScore = pct;
    toast(pct >= 60 ? "Great job! 🎉" : "Keep practising! 💪", pct >= 60 ? "ok" : "warn");
    renderQuizResult();
  }

  function renderQuizResult() {

    else if (pct >= 50) msg = "Good effort! Review the wrong answers and try again.";
    else msg = "Don't worry — every attempt makes you stronger.";

    let html = '<div class="quiz-shell"><div class="result-card">';
    html += '<div class="result-ring" style="--pct:' + pct + '%"><div class="score">' + pct + '<small>%</small></div></div>';
    html += '<h2>' + esc(pct >= 60 ? t("wellDone") : t("keepGoing")) + '</h2>';
    html += '<p class="msg">' + msg + '</p>';
    html += '<div class="result-stats">';
    html += '<div class="rs"><div class="v">' + correct + '</div><div class="l">' + esc(t("correct")) + '</div></div>';
    html += '<div class="rs"><div class="v">' + (total - correct) + '</div><div class="l">' + esc(t("wrong")) + '</div></div>';

    html += '<div class="review-list"><div class="qs-label" style="margin-bottom:10px;">' + esc(t("review")) + '</div>';
    q.questions.forEach(function (cur, i) {
      const chosen = q.answers[i];
      const isCorrect = chosen === cur.a;
      html += '<div class="review-item"><div class="rq">' + (i + 1) + '. ' + esc(cur.q) + '</div>';
      if (isCorrect) html += '<div>✓ <span class="ra">' + esc(cur.o[cur.a]) + '</span></div>';

    html += '<div class="quiz-foot" style="justify-content:center;margin-top:26px;">';
    html += '<button class="btn btn-primary" id="qRetry">' + esc(t("tryAgain")) + '</button>';
    html += '<button class="btn btn-ghost" id="qAnother">' + esc(t("newQuiz")) + '</button>';
    html += '</div></div></div>';

  }

  function startFlashcards(subjId, chId) {
    const s = findSubject(subjId);
    if (!s) return;
    let cards = [];
    if (chId) { const c = findChapter(subjId, chId); if (c) cards = c.c.slice(); }
    else { s.chapters.forEach(function (c) { cards = cards.concat(c.c); }); }
    if (!cards.length) { toast("No flashcards for this yet.", "warn"); return; }
    shuffle(cards);
    state.fc = { active: true, subjectId: subjId, subjectName: s.name, cards: cards, index: 0, flipped: false };
    setView("flashcards");
  }

    if (/exam prep|exam tips|how to prepare|exam strategy/.test(lower)) {

    }

    if (/generate set|study set|practice set/.test(lower)) {
      return "📊 **To generate a study set:**\n\nScroll up and use the **Study Set Generator** card. Pick a subject, chapter, and how many questions you want!";
    }

    /* Explain X */
    if (/^explain\s+/.test(lower)) {
      const topic = lower.replace(/^explain\s+/, "").trim();
      const found = findKB(topic);
      if (found) return found[lang] || found.en;
    }

    /* Formula */
    if (/formula/.test(lower)) {
      return "📐 **Key formulas for Class " + state.classLevel + ":**\n\nGo to the **Formulas** tab in the sidebar to see all formulas with examples. Or ask me about a specific topic!";
    }

    /* Solve */
    if (/solve|problem|homework/.test(lower)) {
      return "🧮 **I'd love to help solve that!**\n\nPlease type the **full problem**, e.g.:\n• \"Solve 2x + 5 = 15\"\n• \"Find HCF of 12 and 18\"";
    }

    /* Greetings / thanks / help */
    if (/^(hi|hello|hey|namaste|ഹായ്|ഹലോ|مرحبا|السلام)/i.test(lower)) return t("greeting");
    if (/(thank|thanks|നന്ദി|شكرا)/i.test(lower)) return t("thanks");
    if (/^(help|what can you|how do i use|സഹായം|مساعدة)/i.test(lower)) return t("helpMsg");

    /* KB */
    const found = findKB(lower);
    if (found) return found[lang] || found.en;

    /* Fallback: search quiz questions */
    const subs = subjects();
    for (let i = 0; i < subs.length; i++) {
      const s = subs[i];
      for (let j = 0; j < s.chapters.length; j++) {
        const c = s.chapters[j];
        for (let k = 0; k < c.q.length; k++) {
          const cur = c.q[k];
          const qLower = cur.q.toLowerCase();
          const words = lower.split(/\s+/).filter(function (w) { return w.length > 3; });
          let matchCount = 0;
          words.forEach(function (w) { if (qLower.indexOf(w) !== -1) matchCount++; });
          if (words.length > 0 && matchCount / words.length >= 0.5) {
            return "**" + cur.q + "**\n\n✅ **Answer:** " + cur.o[cur.a] + "\n\n💡 " + (cur.e || cur.h || "");
          }
        }
      }
    }

    return t("noMatch");
  }

  function findKB(query) {
    let best = null, bestScore = 0;
    for (let i = 0; i < KB.length; i++) {
      const entry = KB[i];
      let score = 0;
      for (let j = 0; j < entry.keys.length; j++) {
        const k = entry.keys[j].toLowerCase();
        if (query.indexOf(k) !== -1) score += k.length;
      }
      if (score > bestScore) { bestScore = score; best = entry; }
    }
    return bestScore > 0 ? best : null;
  }

  function setTimerMode(mode) {
    const tm = state.timer;
    tm.running = false;
    if (tm.interval) { clearInterval(tm.interval); tm.interval = null; }
    tm.mode = mode;
    if (mode === "focus") tm.total = 25 * 60;
    else if (mode === "short") tm.total = 5 * 60;
    else tm.total = 15 * 60;
    tm.seconds = tm.total;
    renderTimer();
  }

  function toggleTimer() {
    const tm = state.timer;
    if (tm.running) {
      tm.running = false;
      if (tm.interval) { clearInterval(tm.interval); tm.interval = null; }
      renderTimer(); return;
    }
    tm.running = true;
    tm.interval = setInterval(function () {
      tm.seconds--;
      if (tm.seconds <= 0) {
        clearInterval(tm.interval); tm.interval = null; tm.running = false;
        if (tm.mode === "focus") tm.sessions++;
        toast(tm.mode === "focus" ? "Focus session done! 🎉" : "Break over!", "ok");
        try {
          if (window.AudioContext || window.webkitAudioContext) {
            const ctx = new (window.AudioContext || window.webkitAudioContext)();
            const o = ctx.createOscillator(), g = ctx.createGain();
            o.connect(g); g.connect(ctx.destination);
            o.frequency.value = 660; g.gain.value = 0.08;
            o.start(); setTimeout(function () { o.stop(); }, 260);
          }
        } catch (e) {}
        tm.seconds = tm.total; renderTimer(); return;
      }
      const ring = view.querySelector(".timer-ring");
      const timeEl = view.querySelector(".time");
      if (timeEl) timeEl.textContent = pad(Math.floor(tm.seconds / 60)) + ":" + pad(tm.seconds % 60);
      if (ring) ring.style.setProperty("--tpct", (100 - (tm.seconds / tm.total) * 100) + "%");
    }, 1000);
    renderTimer();
  }

  function resetTimer() {
    const tm = state.timer;
    tm.running = false;
    if (tm.interval) { clearInterval(tm.interval); tm.interval = null; }
    tm.seconds = tm.total;
    renderTimer();
  }

      html += '<button class="btn btn-primary" id="noteSave">' + esc(t("save")) + '</button>';
      html += '<button class="btn btn-ghost" id="noteDelete">' + esc(t("delete")) + '</button></div>';
    } else {
      html += '<div class="empty"><div class="em">📝</div><h3>' + esc(t("noNotes")) + '</h3><p>' + esc(t("noNotesDesc")) + '</p></div>';
    }
    html += '</div></div>';
    view.innerHTML = html;

    const newBtn = view.querySelector("#noteNew");
    if (newBtn) newBtn.addEventListener("click", function () {
      const id = "n" + Date.now();
      state.notes.unshift({ id: id, title: "New Note", body: "", updated: Date.now() });
      state.activeNoteId = id; saveNotes(); renderNotes();
      setTimeout(function () { const tt = view.querySelector("#noteTitle"); if (tt) { tt.focus(); tt.select(); } }, 40);
    });
    view.querySelectorAll("[data-note]").forEach(function (b) {
      b.addEventListener("click", function () { state.activeNoteId = b.getAttribute("data-note"); renderNotes(); });
    });
    const saveBtn = view.querySelector("#noteSave");
    if (saveBtn) saveBtn.addEventListener("click", function () {
      const tt = view.querySelector("#noteTitle").value.trim() || "Untitled";
      const bb = view.querySelector("#noteBody").value;
      const note = state.notes.find(function (n) { return n.id === state.activeNoteId; });
      if (note) { note.title = tt; note.body = bb; note.updated = Date.now(); saveNotes(); toast("Note saved ✓", "ok"); renderNotes(); }
    });
    const delBtn = view.querySelector("#noteDelete");
    if (delBtn) delBtn.addEventListener("click", function () {
      if (!confirm("Delete this note?")) return;
      state.notes = state.notes.filter(function (n) { return n.id !== state.activeNoteId; });
      state.activeNoteId = state.notes.length ? state.notes[0].id : null;
      saveNotes(); renderNotes(); toast("Note deleted", "warn");
    });
  }

  /* PROGRESS */
  function renderProgress() {
    const subs = subjects();
    const subjectStats = subs.map(function (s) {
      return { id: s.id, name: s.name, emoji: s.emoji, pct: Math.round(subjectProgress(s.id) * 100) };
    });
    const avg = subjectStats.length ? Math.round(subjectStats.reduce(function (a, b) { return a + b.pct; }, 0) / subjectStats.length) : 0;
    const classQuizzes = state.progress.quizzes.filter(function (q) { return !q.classLevel || q.classLevel === state.classLevel; });
    const quizCount = classQuizzes.length;

    const days = [];
    const today = new Date();
    for (let i = 27; i >= 0; i--) {
      const d = new Date(today.getTime() - i * 86400000);
      const key = d.toISOString().slice(0, 10);
      days.push(state.progress.activity[key] || 0);
    }

    let html = '<div class="page-head"><h1>' + esc(t("progress")) + '</h1><p>' + esc(t("progressDesc")) + '</p></div>';
    html += '<div class="prog-grid">';
    html += '<div class="prog-card"><h4>' + esc(t("overallMastery")) + '</h4><div style="text-align:center;padding:6px 0 4px;">';
    html += '<div style="font-size:2.4rem;font-weight:800;">' + avg + '%</div>';
    html += '<div style="font-size:0.8rem;color:var(--text-muted);margin-top:2px;font-weight:600;">' + esc(t("acrossAll")) + '</div></div></div>';
    html += '<div class="prog-card"><h4>' + esc(t("quizzesTaken")) + '</h4><div style="text-align:center;padding:6px 0 4px;">';
    html += '<div style="font-size:2.4rem;font-weight:800;">' + quizCount + '</div>';
    html += '<div style="font-size:0.8rem;color:var(--text-muted);margin-top:2px;font-weight:600;">' + esc(t("keepGoingMsg")) + '</div></div></div>';
    html += '<div class="prog-card"><h4>' + esc(t("studyStreak")) + '</h4><div style="text-align:center;padding:6px 0 4px;">';
    html += '<div style="font-size:2.4rem;font-weight:800;">🔥 ' + state.progress.streak.count + '</div>';
    html += '<div style="font-size:0.8rem;color:var(--text-muted);margin-top:2px;font-weight:600;">' + esc(state.progress.streak.count === 1 ? t("day") : t("days")) + '</div></div></div>';
    html += '</div>';

    html += '<div class="prog-card" style="margin-bottom:16px;"><h4>' + esc(t("subjectMastery")) + '</h4>';
    subjectStats.forEach(function (st) {
      html += '<div class="bar-row"><div class="nm">' + st.emoji + ' ' + esc(st.name) + '</div>';
      html += '<div class="track"><div class="fill" style="width:' + st.pct + '%;background:var(--brand);"></div></div>';
      html += '<div class="pct">' + st.pct + '%</div></div>';
    });
    html += '</div>';

    html += '<div class="prog-card"><h4>' + esc(t("last4Weeks")) + '</h4><div class="heat">';
    days.forEach(function (n) {
      let lvl = "";
      if (n > 0) lvl = "l1";
      if (n >= 2) lvl = "l2";
      if (n >= 4) lvl = "l3";
      if (n >= 6) lvl = "l4";
      html += '<div class="day ' + lvl + '"></div>';
    });
    html += '</div>';
    html += '<div style="margin-top:12px;font-size:0.75rem;color:var(--text-muted);font-weight:600;">' + esc(t("heatDesc")) + '</div></div>';

    if (classQuizzes.length) {
      html += '<div class="prog-card" style="margin-top:16px;"><h4>' + esc(t("recentQuizzes")) + '</h4>';
      classQuizzes.slice(-6).reverse().forEach(function (q) {
        html += '<div class="bar-row"><div class="nm" style="width:auto;flex:1;">' + esc(q.subjectName) + ' · ' + esc(q.chapterName) + '</div>';
        html += '<div class="pct" style="width:auto;color:' + (q.score >= 60 ? "var(--ok)" : "var(--warn)") + ';">' + q.score + '%</div></div>';
      });
      html += '</div>';
    }
    view.innerHTML = html;
  }

  /* SEARCH */
  $("searchInput").addEventListener("input", function (e) {
    const q = e.target.value.trim().toLowerCase();
    if (!q) return;
    const subs = subjects();
    for (let i = 0; i < subs.length; i++) {
      const s = subs[i];
      if (s.name.toLowerCase().indexOf(q) !== -1) { setView("subjects", { subjectId: s.id }); return; }
      for (let j = 0; j < s.chapters.length; j++) {
        const c = s.chapters[j];
        if (c.name.toLowerCase().indexOf(q) !== -1 || c.desc.toLowerCase().indexOf(q) !== -1) {
          setView("subjects", { subjectId: s.id }); return;
        }
      }
    }
  });

  }
